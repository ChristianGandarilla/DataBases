## Principal commands for SQL

C = CREATE - CREATE TABLE table_name (
    column1 datatype constraints,
    column2 datatype constraints,
    column3 datatype constraints,
    ... );
R = READ - SELECT columns FROM table_name WHERE conditions;
U = UPDATE - UPDATE table_name SET column1 = value1, column2 = value2, ... WHERE condition;
D = DELETE - DELETE FROM table_name WHERE condition;.
